\begin{frame}{Notation}
  \begin{itemize}
      \item<1-> \(W^{[l]}\) -- weight matrix on layer \(l\)
      \item<2-> \(b^{[l]}\) -- bias vector on layer \(l\)
      \item<3-> \(z^{[l]}_i \only<4>{\textcolor{orange}{= W^{[l]} a^{[l-1]} + b^{[l]}}}\) -- weighted input of the \(i\textsuperscript{th}\) neuron on layer \(l\)
      \item<5-> \(a^{[l]}_i\only<6>{\textcolor{orange}{= a(z^{[l]})}}\) -- output of the \(i\textsuperscript{th}\)
          neuron on layer \(l\) \only<6>{\textcolor{orange}{(\(a\) activation function)}}
      \item<7-> \(y(x^{\{i\}})\) -- \(i\textsuperscript{th}\) observed data (input \(x^{\{i\}}\), and corresponding output \(y(x^{\{i\}})\))
  \end{itemize}
\end{frame}

\begin{frame}
    \begin{itemize}
        \item Fully connected \(2\)--\(3\)--\(2\) \ac{NN} Forward Pass
           \[F(x) = \sigma\!\left( W^{[4]} \;\sigma\!\left(W^{[3]}\; \sigma\!\left(W^{[2]} \; x +b^{[2]}\right) + b^{[3]} \right)
           + b^{[4]}\right)\]
           \pause\item \(L^{2}\) Loss (for observations \(\{y(x^{\{i\}})\mid i \in \hat{N}\}\))
            \[
                \ell_2(W^{[2]}, W^{[3]}, W^{[4]}, b^{[2]}, b^{[3]}, b^{[4]}) = \frac{1}{2N}\sum_{i
                = 1}^{N} \|y(x^{\{i\}}) - F(x^{\{i\}})\|_2^2.
            \] 
    \end{itemize}
\end{frame}

\section{Back Propagation}%
\label{sec:back-propagation}


\begin{frame}
    \begin{itemize}
        \item Sensitivity \(\delta^{[l]}_j = \pdv{C}{z_j^{[l]}}\)
            \[
                \delta^{[L]} = a'(z^{[L]}) \cdot (a^{[L]} - y) 
            \] 
            \[
                \delta^{[l]} = a'(z^{[l]}) \cdot (W^{[l+1]})^T \delta^{[l+1]}
            \] 
        \pause\item Gradient \(\nabla_W C\) \& \(\nabla_b C\)
            \[
                \pdv{C}{b^{[l]_j}}  = \delta^{[l]}_j
            \] 
            \[
                \pdv{C}{W^{[l]}_{ij}} = \delta^{[l]}_i \cdot a^{[l-1]}_j
            \] 
    \end{itemize}
\end{frame}


\begin{frame}[fragile]
    \begin{minted}[breaklines,escapeinside=||,mathescape=true, numbersep=3pt, gobble=2, frame=single,
        fontsize=\fontsize{6pt}{7pt}\selectfont, framesep=2mm]{julia1}
function train_NN_3_2_3(x, y; η=0.05, iterations=Int(1e6))
    global W2, b2, W3, b3, W4, b4 # assuming that the parameters are initialized

    for _ in Base.OneTo(iterations)
        ind = rand(Base.OneTo(size(x, 2))) # choose observation for gradient computation
        x_train = x[:, ind]
        y_train = y[:, ind]

        # Forward pass
        a2 = σ.(W2 * x_train .+ b2)
        a3 = σ.(W3 * a2 .+ b3)
        a4 = σ.(W4 * a3 .+ b4) # a4

        # Backward pass
        δ4 = @. a4 * (1 - a4) * (a4 - y_train)  # using σ'(x) = σ(x)(1 - σ(x))
        δ3 = a3 .* (1 .- a3) .* (W4' * δ4)
        δ2 = a2 .* (1 .- a2) .* (W3' * δ3)

        # Step gradient
        W2 .-= η * δ2 * x_train'
        W3 .-= η * δ3 * a2'
        W4 .-= η * δ4 * a3'
        b2 .-= η * δ2
        b3 .-= η * δ3
        b4 .-= η * δ4
    end

    return W2,b2,W3,b3,W4,b4
end
    \end{minted} 
\end{frame}
